#!/usr/bin/env python3
"""N0160 focused daemon CPU reproduction; run inside the private dev-1 lab."""
import argparse
import csv
import json
import os
from pathlib import Path
import subprocess
import time


def proc_table():
    rows = {}
    for entry in Path('/proc').iterdir():
        if not entry.name.isdigit():
            continue
        try:
            raw = (entry / 'stat').read_text()
            tail = raw[raw.rfind(')') + 2:].split()
            pid = int(entry.name)
            rows[pid] = (int(tail[1]), int(tail[11]) + int(tail[12]),
                         int(tail[13]) + int(tail[14]))
        except (OSError, ValueError, IndexError):
            continue
    return rows


def process_tree(root, table):
    found = {root}
    while True:
        children = {pid for pid, row in table.items() if row[0] in found}
        if children <= found:
            return found
        found.update(children)


def thread_ticks(pid):
    values = {}
    for entry in (Path('/proc') / str(pid) / 'task').glob('*/stat'):
        try:
            raw = entry.read_text()
            tail = raw[raw.rfind(')') + 2:].split()
            values[int(entry.parent.name)] = int(tail[11]) + int(tail[12])
        except (OSError, ValueError, IndexError):
            continue
    return values


def command(args, *argv):
    subprocess.run([str(args.nacrectl), *map(str, argv)], env=os.environ.copy(),
                   check=True, stdout=subprocess.DEVNULL)


def dispatch(args, *parts):
    command(args, 'ui', *parts)


def interaction(args):
    for cycle in range(6):
        start = time.monotonic()
        dispatch(args, 'nacre', 'launcher', '')
        time.sleep(.8)
        dispatch(args, 'nacre', 'close')
        dispatch(args, 'drawers', 'toggle', 'dashboard')
        dispatch(args, 'nacre', 'tab', '0')
        time.sleep(.8)
        dispatch(args, 'nacre', 'close')
        for workspace in ('2', '1', '3', '1'):
            subprocess.run(['niri', 'msg', 'action', 'focus-workspace', workspace], check=True,
                           stdout=subprocess.DEVNULL)
            deadline = time.monotonic() + 3
            active = None
            while time.monotonic() < deadline:
                result = subprocess.run(['niri', 'msg', '-j', 'workspaces'], check=True,
                                        text=True, capture_output=True)
                rows = json.loads(result.stdout)
                if isinstance(rows, dict):
                    rows = rows.get('workspaces', [])
                current = next((row for row in rows if row.get('is_focused') or row.get('focused')), None)
                if current:
                    active = current.get('idx', current.get('index'))
                    if str(active) == workspace:
                        break
                time.sleep(.05)
            if str(active) != workspace:
                raise RuntimeError(f'workspace switch to {workspace} not observed (active={active})')
            time.sleep(.3)
        for index in range(4):
            subprocess.run(['gdbus', 'call', '--session', '--dest', 'org.freedesktop.Notifications',
                            '--object-path', '/org/freedesktop/Notifications', '--method',
                            'org.freedesktop.Notifications.Notify', 'N0154', '0', '',
                            f'Benchmark {cycle + 1}.{index + 1}', 'Identical private notification burst',
                            '[]', '{}', '2500'], check=True, stdout=subprocess.DEVNULL)
            time.sleep(.2)
        time.sleep(max(0, start + 10 - time.monotonic()))


def sample(args, phase, duration):
    hz = os.sysconf('SC_CLK_TCK')
    fields = ['phase', 'second', 'daemon_cpu_pct', 'daemon_self_cpu_pct',
              'daemon_waited_child_pct', 'thread_cpu_pct_by_tid_json',
              'tree_cpu_pct', 'process_cpu_pct_by_pid_json',
              'process_cmdline_by_pid_json', 'tree_processes', 'sample_elapsed_s',
              'daemon_delta_ticks', 'daemon_self_delta_ticks', 'daemon_waited_delta_ticks',
              'tree_delta_ticks', 'mem_available_kib']
    with args.out.open('a', newline='') as stream:
        writer = csv.DictWriter(stream, fieldnames=fields)
        if stream.tell() == 0:
            writer.writeheader()
        def snapshot():
            table = proc_table()
            tree = process_tree(args.daemon_pid, table)
            daemon_row = table.get(args.daemon_pid, (None, 0, 0))
            threads = thread_ticks(args.daemon_pid)
            daemon_self = daemon_row[1]
            daemon_waited = daemon_row[2]
            daemon = daemon_self + daemon_waited
            processes = {}
            commands = {}
            for pid in tree:
                row = table.get(pid, (None, 0, 0))
                processes[pid] = (row[1], row[2])
                try:
                    raw = (Path('/proc') / str(pid) / 'cmdline').read_bytes()
                    commands[str(pid)] = raw.replace(b'\0', b' ').decode(errors='replace').strip()
                except OSError:
                    commands[str(pid)] = ''
            total = sum(own + waited for own, waited in processes.values())
            return time.monotonic(), daemon, daemon_self, daemon_waited, threads, processes, total, commands

        previous = snapshot()
        start = previous[0]
        for second in range(1, duration + 1):
            target = start + second
            while time.monotonic() < target:
                time.sleep(max(0, min(.05, target - time.monotonic())))
            now, daemon, daemon_self, daemon_waited, threads, processes, total, commands = snapshot()
            mem_available = int(next(line for line in Path('/proc/meminfo').read_text().splitlines()
                                     if line.startswith('MemAvailable:')).split()[1])
            if mem_available < 2 * 1024 * 1024:
                raise RuntimeError('MemAvailable below 2 GiB capture safety gate')
            if previous:
                elapsed = now - previous[0]
                thread_delta = {str(tid): round((ticks - previous[4].get(tid, ticks)) * 100 / hz / elapsed, 3)
                                for tid, ticks in threads.items()}
                process_delta = {}
                for pid, (self_ticks, child_ticks) in processes.items():
                    old_self, old_child = previous[5].get(pid, (self_ticks, child_ticks))
                    self_pct = round((self_ticks - old_self) * 100 / hz / elapsed, 3)
                    child_pct = round((child_ticks - old_child) * 100 / hz / elapsed, 3)
                    process_delta[str(pid)] = {'self_pct': self_pct, 'waited_child_pct': child_pct,
                                               'inclusive_pct': round(self_pct + child_pct, 3)}
                writer.writerow({'phase': phase, 'second': second,
                    'daemon_cpu_pct': round((daemon - previous[1]) * 100 / hz / elapsed, 3),
                    'daemon_self_cpu_pct': round((daemon_self - previous[2]) * 100 / hz / elapsed, 3),
                    'daemon_waited_child_pct': round((daemon_waited - previous[3]) * 100 / hz / elapsed, 3),
                    'thread_cpu_pct_by_tid_json': json.dumps(thread_delta, sort_keys=True),
                    'tree_cpu_pct': round((total - previous[6]) * 100 / hz / elapsed, 3),
                    'process_cpu_pct_by_pid_json': json.dumps(process_delta, sort_keys=True),
                    'process_cmdline_by_pid_json': json.dumps(commands, sort_keys=True),
                    'tree_processes': len(processes),
                    'sample_elapsed_s': now - previous[0],
                    'daemon_delta_ticks': daemon - previous[1],
                    'daemon_self_delta_ticks': daemon_self - previous[2],
                    'daemon_waited_delta_ticks': daemon_waited - previous[3],
                    'tree_delta_ticks': total - previous[6],
                    'mem_available_kib': mem_available})
                stream.flush()
            previous = (now, daemon, daemon_self, daemon_waited, threads, processes, total, commands)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--daemon-pid', type=int, required=True)
    parser.add_argument('--nacrectl', type=Path, required=True)
    parser.add_argument('--out', type=Path, required=True)
    parser.add_argument('--phase', choices=('idle', 'interaction'), required=True)
    parser.add_argument('--seconds', type=int, default=60)
    args = parser.parse_args()
    if args.phase == 'interaction':
        # Start the shared six-cycle schedule alongside one-second accounting.
        import threading
        failure = []
        def workload():
            try:
                interaction(args)
            except BaseException as error:
                failure.append(error)
        worker = threading.Thread(target=workload)
        worker.start()
        sample(args, args.phase, args.seconds)
        worker.join()
        if failure:
            raise failure[0]
    else:
        sample(args, args.phase, args.seconds)


if __name__ == '__main__':
    main()
