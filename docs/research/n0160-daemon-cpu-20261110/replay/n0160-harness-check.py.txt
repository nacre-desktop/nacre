"""Run in the lab container: validate replay timing and group CPU accounting."""
import csv
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
import tempfile
from types import SimpleNamespace
from unittest.mock import patch


def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value


here = Path(__file__).parent
runner = module('runner', here / 'n0160-run.py')
clock = [0.0]
actions, notifications, dispatches = [], [], []


class Arm:
    active = 1

    def dispatch(self, *args):
        dispatches.append((clock[0], args))

    def run(self, command, **kwargs):
        if command[:4] == ['niri', 'msg', 'action', 'focus-workspace']:
            self.active = int(command[4])
            actions.append((clock[0], self.active))
        elif command[:4] == ['niri', 'msg', '-j', 'workspaces']:
            return SimpleNamespace(stdout=json.dumps([{'idx': self.active, 'is_focused': True}]))
        elif command[0] == 'gdbus':
            notifications.append(clock[0])
        else:
            raise AssertionError(command)
        return SimpleNamespace(returncode=0, stdout='', stderr='')


with patch.object(runner.time, 'monotonic', side_effect=lambda: clock[0]), patch.object(
        runner.time, 'sleep', side_effect=lambda seconds: clock.__setitem__(0, clock[0] + seconds)):
    runner.interaction(Arm())
assert clock[0] == 60.0, clock
assert len(dispatches) == 30 and len(actions) == 24 and len(notifications) == 24
for cycle in range(6):
    actual = [round(t - 10 * cycle, 6) for t, _ in actions[cycle * 4:cycle * 4 + 4]]
    assert actual == [1.6, 1.9, 2.2, 2.5], actual
    assert [v for _, v in actions[cycle * 4:cycle * 4 + 4]] == [2, 1, 3, 1]
    actual = [round(t - 10 * cycle, 6) for t in notifications[cycle * 4:cycle * 4 + 4]]
    assert actual == [2.8, 3.0, 3.2, 3.4], actual
print('Replay: six exact 10s cycles; 30 UI calls, 24 spaced workspace actions, 24 notices: PASS')

sampler = module('sampler', here / 'n0160-repro.py')
with tempfile.TemporaryDirectory(prefix='n0160-sampler-', dir='/tmp') as root:
    root = Path(root)
    child = subprocess.Popen([sys.executable, '-c', '''
import pathlib, sys, threading, time
pathlib.Path(sys.argv[1]).write_text('ready')
while not pathlib.Path(sys.argv[2]).exists(): time.sleep(.005)
def work():
    until = time.thread_time() + .3
    while time.thread_time() < until: pass
t = threading.Thread(target=work); t.start(); t.join(); time.sleep(5)
''', str(root / 'ready'), str(root / 'go')])
    try:
        import time
        while not (root / 'ready').exists():
            time.sleep(.005)
        trigger = __import__('threading').Timer(.2, lambda: (root / 'go').touch())
        trigger.start()
        sampler.sample(SimpleNamespace(daemon_pid=child.pid, out=root / 'cpu.csv'), 'smoke', 2)
        trigger.join()
        rows = list(csv.DictReader((root / 'cpu.csv').open()))
        assert len(rows) == 2
        ticks = sum(int(row['daemon_self_delta_ticks']) for row in rows)
        assert 20 <= ticks <= 60, ticks
        assert all(float(row['sample_elapsed_s']) > 0 for row in rows)
        print(f'Sampler: captured exited worker CPU, {ticks} group ticks in two full deltas: PASS')
    finally:
        child.terminate()
        child.wait(timeout=5)
