"""Five alternating private N0160 before/after pairs, inside n0160-lab only."""
import argparse
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import time

parser = argparse.ArgumentParser()
parser.add_argument('--start', type=int, default=1)
parser.add_argument('--end', type=int, default=5)
parser.add_argument('--first-arm', choices=('before', 'after'), default='before')
args = parser.parse_args()
if not Path('/.dockerenv').exists():
    raise RuntimeError('Run in task-private n0160-lab')
here = Path(__file__).parent
output = Path('/work/.lead/tmp/n0160/pairs')
output.mkdir(parents=True, exist_ok=True)
files = ('cpu.csv', 'memory.csv', 'artifacts.json', 'renderer.json', 'interaction.json',
         'raster-cache.jsonl', 'ui-errors.json', 'daemon.pid', 'daemon.log', 'shell.log', 'niri.log',
         'labwc.log', 'compositor-ready.json', 'lab-complete')
for pair in range(args.start, args.end + 1):
    for arm in ('before', 'after'):
        if pair == args.start and args.first_arm == 'after' and arm == 'before':
            continue
        name = f'pair{pair:02}-{arm}'
        destination = output / name
        private = Path('/tmp') / f'n0160-{name}'
        if destination.exists() or private.exists():
            raise RuntimeError(f'Refusing overwrite of capture {name}')
        destination.mkdir(mode=0o700)
        helper = here / 'tmp' / ('n0160-baseline-helper' if arm == 'before' else 'n0160-candidate-helper')
        command = [sys.executable, str(here / 'n0160-run.py'), '--repo', '/work',
                   '--helper', str(helper), '--shell', '/work/target/release/nacre-shell',
                   '--bin', '/work/target/release', '--out', str(private),
                   '--warmup-seconds', '30', '--idle-seconds', '30',
                   '--interaction-seconds', '60', '--post-idle-seconds', '30']
        started = time.monotonic()
        print(f'START {name}', flush=True)
        with (destination / 'run.log').open('w') as log:
            result = subprocess.run(command, stdout=log, stderr=subprocess.STDOUT)
        for filename in files:
            if (private / filename).is_file():
                shutil.copy2(private / filename, destination / filename)
        geometry = private / 'bead/geometry.json'
        if geometry.is_file():
            shutil.copy2(geometry, destination / 'geometry.json')
        (destination / 'result.json').write_text(json.dumps({
            'exit_code': result.returncode, 'elapsed_s': time.monotonic() - started,
            'pair': pair, 'arm': arm, 'command': command,
        }, indent=2) + '\n')
        if private.exists():
            shutil.rmtree(private)
        print(f'END {name} rc={result.returncode}', flush=True)
        if result.returncode:
            raise RuntimeError(f'Capture {name} failed; preserve evidence and inspect before retry')
