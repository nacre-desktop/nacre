#!/usr/bin/env bash
# Replicates the complete `desktop` job from .github/workflows/check.yml.
# Run inside n0160-lab from /work after installing its listed Arch packages.
set -euo pipefail
cd /work
nice -n 10 pacman -Syu --noconfirm --ask=4 --needed git curl unzip python python-pip python-pillow ruff lua fish qt6-declarative qt6-imageformats gcc pkgconf rust clang libpipewire mpv lcms2 cmake make gtk3 librsvg kconfig kcolorscheme breeze nodejs npm jq xorg-server-xvfb xorg-xauth kitty less mesa dbus ttf-dejavu niri sway swaybg grim fastfetch ffmpeg gdk-pixbuf2 curl unzip fontconfig libxkbcommon wayland
export RUNNER_TEMP=/tmp/runner-temp
export CI=true
mkdir -p "$RUNNER_TEMP"
nice -n 10 python tools/ci_public_checkout.py
nice -n 10 cargo build -j2 --release --locked --manifest-path deps/orient/Cargo.toml -p orient-cli
nice -n 10 python -m venv /tmp/check-env
nice -n 10 python - <<PY
import json, subprocess
pins = json.load(open("dependencies.lock.json"))["python"]
subprocess.run(["/tmp/check-env/bin/pip", "install", "./nacre/shell-cli", "ruff==0.16.10", "dbus-next==0.2.3", *[name + "==" + version for name, version in pins.items()]], check=True)
PY
export RUST_TEST_THREADS=2
export BEAD_TEST_ARTIFACTS="$RUNNER_TEMP/bead-headless"
nice -n 10 cargo fmt --check
nice -n 10 cargo clippy --workspace --all-targets --locked -j2 -- -D warnings
nice -n 10 cargo build -j2 --locked --examples
nice -n 10 cargo test --workspace --locked -j2
nice -n 10 cargo run -j2 --locked --manifest-path crates/nacre-protocol/tests/minimal-consumer/Cargo.toml --target-dir target
nice -n 10 cargo build -j2 --locked --release -p nacre-helper --bin nacre-helper
unset RUST_TEST_THREADS BEAD_TEST_ARTIFACTS
nice -n 10 cargo test -p nacre-shell --features trace --locked -j2 memory_policy::tests::
nice -n 10 cargo check -p nacre-shell --no-default-features --locked -j2
nice -n 10 cargo check --workspace --all-features --locked -j2
nice -n 10 cargo build -p nacre-shell --bin nacre-shell --no-default-features --locked -j2
nice -n 10 cargo build -p nacre-shell --bin bead-wallpaper-video --locked -j2
nice -n 10 cargo build -p nacre-shell --bin nacre-wallpaper-decode --locked -j2
nice -n 10 python tools/check_product_shell.py --download-analyzer "$RUNNER_TEMP/luau/luau-analyze"
nice -n 10 python tools/check_product_shell.py --analyzer "$RUNNER_TEMP/luau/luau-analyze"
asset=cargo-deny-0.20.2-x86_64-unknown-linux-musl.tar.gz
base=https://github.com/EmbarkStudios/cargo-deny/releases/download/0.20.2
nice -n 10 curl -fsSL "$base/$asset" -o "$RUNNER_TEMP/$asset"
nice -n 10 curl -fsSL "$base/$asset.sha256" -o "$RUNNER_TEMP/deny.sha256"
cd "$RUNNER_TEMP"
printf '%s  %s\n' "$(cat deny.sha256)" "$asset" | sha256sum -c -
tar -xzf "$asset"
mkdir -p "$HOME/.cargo/bin"
install -m 755 "$RUNNER_TEMP/${asset%.tar.gz}/cargo-deny" "$HOME/.cargo/bin/cargo-deny"
export PATH="$HOME/.cargo/bin:$PATH"
cd /work
nice -n 10 cargo deny --manifest-path crates/nacre-shell/Cargo.toml --all-features --config crates/nacre-shell/deny.toml check
nice -n 10 npm ci --ignore-scripts --no-audit --no-fund
nice -n 10 /tmp/check-env/bin/python tools/i18n/check.py
nice -n 10 /tmp/check-env/bin/python tools/i18n/generate.py
export NACRE_ORIENT_BINARY="$PWD/deps/orient/target/release/orient"
export NACRE_HELPER_BINARY="$PWD/target/release/nacre-helper"
nice -n 10 dbus-run-session -- xvfb-run -a /tmp/check-env/bin/python tools/check.py
nice -n 10 env NACRE_HELPER_BINARY="$PWD/target/release/nacre-helper" /tmp/check-env/bin/python tools/capture-aur-review.py --output /tmp/nacre-aur-review
nice -n 10 env NACRE_MEMORY_BINARY="$PWD/target/release/nacre-helper" cargo test -j2 --locked -p nacre-helper --test daemon memory:: -- --ignored --nocapture
